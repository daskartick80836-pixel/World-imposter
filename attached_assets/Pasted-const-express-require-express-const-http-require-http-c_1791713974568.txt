const rooms = {};

io.on('connection', (socket) => {
  // Create Room
  socket.on('createRoom', ({ playerName, avatar, category, language }) => {
    const roomCode = Math.random().toString(36).substring(2, 7).toUpperCase();
    const lang = language || 'en';
    const cat = category || 'vehicles';

    rooms[roomCode] = {
      code: roomCode,
      host: socket.id,
      category: cat,
      language: lang,
      players: [{ id: socket.id, name: playerName, avatar: avatar || '🤖', score: 0, isBot: false }],
      gameState: 'LOBBY',
      secretWord: '',
      impostorId: null,
      clues: [],
      votes: {}
    };
    socket.join(roomCode);
    socket.emit('roomCreated', rooms[roomCode]);
  });

  // Join Room
  socket.on('joinRoom', ({ roomCode, playerName, avatar }) => {
    const room = rooms[roomCode];
    if (!room) return socket.emit('errorMsg', 'Room not found!');
    if (room.gameState !== 'LOBBY') return socket.emit('errorMsg', 'Game already in progress!');
    if (room.players.length >= 10) return socket.emit('errorMsg', 'Room is full!');

    room.players.push({ id: socket.id, name: playerName, avatar: avatar || '🐱', score: 0, isBot: false });
    socket.join(roomCode);
    io.to(roomCode).emit('updateRoom', room);
  });

  // Add AI Bot
  socket.on('addBot', ({ roomCode }) => {
    const room = rooms[roomCode];
    if (room && room.host === socket.id && room.players.length < 10) {
      const botId = 'bot_' + Math.random().toString(36).substring(2, 7);
      const avatars = ['🤖', '👾', '🦊', '🐼', '🦁', '🦉'];
      const botName = `AI Bot ${room.players.length + 1}`;
      
      room.players.push({
        id: botId,
        name: botName,
        avatar: avatars[Math.floor(Math.random() * avatars.length)],
        score: 0,
        isBot: true
      });
      io.to(roomCode).emit('updateRoom', room);
    }
  });

  // Start / Next Round Game
  socket.on('startGame', ({ roomCode }) => {
    const room = rooms[roomCode];
    if (!room || room.host !== socket.id) return;
    if (room.players.length < 3) return socket.emit('errorMsg', 'Need at least 3 players to start!');

    const langDb = WORD_DB[room.language] || WORD_DB.en;
    const words = langDb[room.category] || langDb.vehicles;
    room.secretWord = words[Math.floor(Math.random() * words.length)];

    const impostorIndex = Math.floor(Math.random() * room.players.length);
    room.impostorId = room.players[impostorIndex].id;

    room.gameState = 'IN_GAME';
    room.clues = [];
    room.votes = {};

    // Generate clues and votes for AI bots automatically
    room.players.forEach(player => {
      if (player.isBot) {
        const botClueList = BOT_CLUES[room.language] || BOT_CLUES.en;
        const randomClue = botClueList[Math.floor(Math.random() * botClueList.length)];
        room.clues.push({ name: player.name, avatar: player.avatar, clue: randomClue });

        // Bot Vote Logic
        const targets = room.players.filter(p => p.id !== player.id);
        const randomTarget = targets[Math.floor(Math.random() * targets.length)];
        room.votes[player.id] = randomTarget.id;
      } else {
        io.to(player.id).emit('gameStarted', {
          role: player.id === room.impostorId ? 'IMPOSTOR' : 'CREW',
          word: player.id === room.impostorId ? '???' : room.secretWord,
          room
        });
      }
    });

    io.to(roomCode).emit('updateRoom', room);
  });

  // Submit Clue
  socket.on('submitClue', ({ roomCode, clue }) => {
    const room = rooms[roomCode];
    if (!room) return;
    const player = room.players.find(p => p.id === socket.id);
    if (player) {
      room.clues.push({ name: player.name, avatar: player.avatar, clue });
      io.to(roomCode).emit('updateRoom', room);
    }
  });

  // Submit Vote
  socket.on('submitVote', ({ roomCode, targetId }) => {
    const room = rooms[roomCode];
    if (!room) return;
    room.votes[socket.id] = targetId;

    if (Object.keys(room.votes).length >= room.players.length) {
      const voteCounts = {};
      Object.values(room.votes).forEach(v => {
        voteCounts[v] = (voteCounts[v] || 0) + 1;
      });

      let suspectId = null;
      let maxVotes = -1;
      for (const [id, count] of Object.entries(voteCounts)) {
        if (count > maxVotes) {
          maxVotes = count;
          suspectId = id;
        }
      }

      const winner = suspectId === room.impostorId ? 'CREW' : 'IMPOSTOR';
      room.gameState = 'RESULTS';

      // Update Scores
      room.players.forEach(p => {
        if (winner === 'CREW' && p.id !== room.impostorId) p.score += 100;
        if (winner === 'IMPOSTOR' && p.id === room.impostorId) p.score += 200;
      });

      io.to(roomCode).emit('gameOver', {
        winner,
        impostorId: room.impostorId,
        secretWord: room.secretWord,
        room
      });
    }
  });

  socket.on('disconnect', () => {
    for (const [code, room] of Object.entries(rooms)) {
      room.players = room.players.filter(p => p.id !== socket.id);
      if (room.players.length === 0) delete rooms[code];
      else io.to(code).emit('updateRoom', room);
    }
  });
});

const PORT = process.env.PORT || 3000;
server.listen(PORT, () => console.log(`Server live on port ${PORT}`));
